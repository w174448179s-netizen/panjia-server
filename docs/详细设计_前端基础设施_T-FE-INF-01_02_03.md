## 三、T-FE-INF-01：前端工程初始化

### 3.1 仓库建立

```bash
# 克隆 plus-ui 官方仓库
git clone https://gitee.com/JavaLionLi/plus-ui.git panjia-ui
cd panjia-ui

# 配置 upstream
git remote rename origin upstream
git remote add origin git@github.com:w174448179s-netizen/panjia-ui.git

git switch -c main

git push -u origin main --force
 

# 安装依赖（必须用 pnpm）
pnpm install
```

### 3.2 盘家目录骨架创建

```bash
mkdir -p src/views/panjia/{login,salary,employee,commission,attendance,analysis}
mkdir -p src/api/panjia
mkdir -p src/store/modules/panjia-app
mkdir -p src/components/panjia
mkdir -p src/assets/panjia
mkdir -p src/styles
```

### 3.3 pnpm 锁定

```bash
# 确保只有 pnpm-lock.yaml，删除其他锁文件
rm -f package-lock.json yarn.lock

# 在 package.json 中强制 pnpm
# "engines": { "pnpm": ">=9" }
# "packageManager": "pnpm@9.x.x"
```

### 3.4 oxlint / oxfmt 配置

plus-ui 6.x 已从 ESLint/Prettier 迁移至 oxlint/oxfmt，盘家不新增配置，复用上游 `.oxlintrc.json` 和 `.oxfmtrc`。如需业务特例规则，在 `.oxlintrc.panjia.json` 中扩展：

```json
{
  "extends": ["./.oxlintrc.json"],
  "rules": {
    "no-console": "warn"
  }
}
```

### 3.5 CI 上游目录守卫

**脚本 `scripts/check-upstream-clean.sh`：**

```bash
#!/bin/bash
# 检查上游目录是否被盘家业务代码污染

DIRS="src/views/system/ src/layout/ src/api/system/"

for dir in $DIRS; do
  if grep -r "panjia" "$dir" 2>/dev/null; then
    echo "❌ 上游目录被污染：$dir 发现了 panjia 相关引用"
    exit 1
  fi
done

echo "✅ 上游目录干净"
```

**CI 配置（`.github/workflows/ci.yml` 或 `Jenkinsfile`）：**

```yaml
# 前端 CI 检查项
steps:
  - name: Check upstream directories
    run: bash scripts/check-upstream-clean.sh
  - name: Lint
    run: pnpm oxlint
  - name: Format check
    run: pnpm oxfmt --check
  - name: Type check
    run: pnpm vue-tsc --noEmit
  - name: Build
    run: pnpm build
```

## 五、T-FE-INF-03：API 服务层与路由骨架

### 5.1 API 请求封装

复用上游 `src/utils/request.ts`，不修改。业务请求统一在 `src/api/panjia/` 下封装。

**`src/api/panjia/index.ts`（请求实例 + 通用配置）：**

```typescript
import request from '@/utils/request'

// 所有盘家业务接口使用 /api/panjia 前缀
const panjiaRequest = {
  get<T>(url: string, params?: any) {
    return request.get<T>(`/api/panjia${url}`, params)
  },
  post<T>(url: string, data?: any) {
    return request.post<T>(`/api/panjia${url}`, data)
  },
  put<T>(url: string, data?: any) {
    return request.put<T>(`/api/panjia${url}`, data)
  },
  delete<T>(url: string, params?: any) {
    return request.delete<T>(`/api/panjia${url}`, params)
  }
}

export default panjiaRequest
```

**示例 `src/api/panjia/employee.ts`：**

```typescript
import panjiaRequest from './index'
import type { Employee, EmployeeQuery } from './types'

export const employeeApi = {
  // 列表（分页）
  list(params: EmployeeQuery) {
    return panjiaRequest.get<PageResult<Employee>>('/employee/page', params)
  },
  // 详情
  getById(id: string) {
    return panjiaRequest.get<Employee>(`/employee/${id}`)
  },
  // 新建
  create(data: EmployeeForm) {
    return panjiaRequest.post('/employee', data)
  },
  // 更新
  update(id: string, data: EmployeeForm) {
    return panjiaRequest.put(`/employee/${id}`, data)
  },
  // 批量导入
  import(file: File) {
    const formData = new FormData()
    formData.append('file', file)
    return panjiaRequest.post('/employee/import', formData)
  }
}
```

### 5.4 全局状态

**`src/store/modules/panjia-app/index.ts`：**

```typescript
import { defineStore } from 'pinia'

interface PanjiaAppState {
  currentStoreId: string      // 当前用户所属门店 ID
  currentStoreName: string    // 当前用户所属门店名称
  roleType: string            // 角色类型：broker/manager/director/hr/salary
  permissionBits: number      // License 权限位
}

export const usePanjiaAppStore = defineStore('panjia-app', {
  state: (): PanjiaAppState => ({
    currentStoreId: '',
    currentStoreName: '',
    roleType: '',
    permissionBits: 0
  }),
  getters: {
    isBroker: (state) => state.roleType === 'broker',
    isManager: (state) => state.roleType === 'manager',
    isDirector: (state) => state.roleType === 'director',
    storeLabel: (state) => state.currentStoreName || '全部门店'
  }
})
```
